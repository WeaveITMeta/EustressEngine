[11] panel_inventory
# is_panel is decided by rule R, which uses only main.slint text.
# A .slint file is a panel iff main.slint instantiates a component that file
# exports, at a site that is either
#   (D) DOCKED  — the instantiation line matches
#                 root\.(left-tab-index|right-tab-index|bottom-panel-mode|active-tab-type)\s*==
#   (F) FLOATED — the instantiation block assigns BOTH x: and y: at its top level,
#                 i.e. main.slint positions it freely over the viewport instead of
#                 handing it to a layout.
# Modal dialogs satisfy neither: they are gated on root.show-* and set no x/y.
# The twelve files main.slint never instantiates (ai_generation.slint,
# collaboration.slint, cursor_badge.slint, main.slint itself, mention_popup.slint,
# network_panel.slint, numeric_field.slint, service_properties.slint,
# soul_settings.slint, tags_editor.slint, theme.slint, view_selector.slint) are
# not panels.
$ml = Get-Content -LiteralPath 'eustress/crates/engine/ui/slint/main.slint'
foreach ($f in Get-ChildItem 'eustress/crates/engine/ui/slint/*.slint') {
  $comps = @((Select-String -LiteralPath $f.FullName -Pattern '^export component ([A-Za-z0-9_]+)').Matches | ForEach-Object { $_.Groups[1].Value })
  foreach ($c in $comps) {
    for ($i=0; $i -lt $ml.Count; $i++) {
      $t = $ml[$i]
      if ($t -notmatch "(^|[^A-Za-z0-9_])$c\s*:?=?\s*\{" -and $t -notmatch ":\s*$c\s*\{") { continue }
      $indent = ($t -replace '(\s*).*','$1').Length; $hasX=$false; $hasY=$false
      for ($j=$i+1; $j -lt $ml.Count; $j++) {
        $b = $ml[$j]
        if ($b -match '^\s*\}\s*$' -and (($b -replace '(\s*).*','$1').Length -le $indent)) { break }
        if ($b -match '^\s+x:\s') { $hasX=$true }; if ($b -match '^\s+y:\s') { $hasY=$true }
      }
      $docked = $t -match 'root\.(left-tab-index|right-tab-index|bottom-panel-mode|active-tab-type)\s*=='
      "$($f.Name)|$c|$($i+1)|docked=$docked|floated=$($hasX -and $hasY)|$($t.Trim())"
    }
  }
}
